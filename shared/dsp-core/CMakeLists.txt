cmake_minimum_required(VERSION 3.22)
project(dsp_core LANGUAGES CXX)

# Keep this subproject usable by SOURCE_SUBDIR without configuring JUCE.
add_library(dsp_core INTERFACE)
add_library(dsp-core::dsp-core ALIAS dsp_core)
target_compile_features(dsp_core INTERFACE cxx_std_17)
target_include_directories(dsp_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")

# This suite can run headlessly from either instrument or a standalone core
# checkout; future Glaze integrations need no JUCE or Dropforge implementation.
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()
if(BUILD_TESTING)
    add_executable(dsp_core_chaos_tests tests/ChaosTests.cpp)
    target_link_libraries(dsp_core_chaos_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME smooth_chaos COMMAND dsp_core_chaos_tests)
    add_executable(dsp_core_oversampling_tests tests/OversamplingTests.cpp)
    target_link_libraries(dsp_core_oversampling_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME oversampling4x COMMAND dsp_core_oversampling_tests)
    add_executable(dsp_core_parallel_saw_drive_tests tests/ParallelSawDriveTests.cpp)
    target_link_libraries(dsp_core_parallel_saw_drive_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME parallel_saw_drive COMMAND dsp_core_parallel_saw_drive_tests)
    add_executable(dsp_core_wavetable_tests tests/WavetableTests.cpp)
    target_link_libraries(dsp_core_wavetable_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME wavetable_bank COMMAND dsp_core_wavetable_tests)
    add_executable(dsp_core_multiband_tests tests/MultibandDynamicsTests.cpp)
    target_link_libraries(dsp_core_multiband_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME multiband_dynamics COMMAND dsp_core_multiband_tests)
    add_executable(dsp_core_stereo_metrics_tests tests/StereoMetricsTests.cpp)
    target_link_libraries(dsp_core_stereo_metrics_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME stereo_metrics COMMAND dsp_core_stereo_metrics_tests)
    add_executable(dsp_core_master_dynamics_tests tests/MasterDynamicsTests.cpp)
    target_link_libraries(dsp_core_master_dynamics_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME master_dynamics COMMAND dsp_core_master_dynamics_tests)
endif()
