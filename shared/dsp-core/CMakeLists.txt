cmake_minimum_required(VERSION 3.22)
project(dsp_core LANGUAGES CXX)

# Keep this subproject usable by SOURCE_SUBDIR without configuring JUCE.
add_library(dsp_core INTERFACE)
add_library(dsp-core::dsp-core ALIAS dsp_core)
target_compile_features(dsp_core INTERFACE cxx_std_17)
target_include_directories(dsp_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")

# This suite can run headlessly from either instrument or a standalone core
# checkout; future Glaze integrations need no JUCE or Dropforge implementation.
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()
if(BUILD_TESTING)
    add_executable(dsp_core_chaos_tests tests/ChaosTests.cpp)
    target_link_libraries(dsp_core_chaos_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME smooth_chaos COMMAND dsp_core_chaos_tests)
    add_executable(dsp_core_oscillator_tests tests/OscillatorTests.cpp)
    target_link_libraries(dsp_core_oscillator_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME oscillator_phase COMMAND dsp_core_oscillator_tests)
    add_executable(dsp_core_peaking_eq_tests tests/PeakingEqTests.cpp)
    target_link_libraries(dsp_core_peaking_eq_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME peaking_eq COMMAND dsp_core_peaking_eq_tests)
    add_executable(dsp_core_notch_tests tests/NotchTests.cpp)
    target_link_libraries(dsp_core_notch_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME tracked_notch COMMAND dsp_core_notch_tests)
    add_executable(dsp_core_oversampling_tests tests/OversamplingTests.cpp)
    target_link_libraries(dsp_core_oversampling_tests PRIVATE dsp-core::dsp-core)
    add_test(NAME oversampling4x COMMAND dsp_core_oversampling_tests)
    add_executable(dsp_core_drive_chain_tests tests/DriveChainTests.cpp)
